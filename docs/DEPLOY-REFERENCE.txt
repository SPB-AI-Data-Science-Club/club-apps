############ NGINX RATE-LIMIT ZONES (/etc/nginx/nginx.conf, http block) ############
# REQUIRED. The server blocks below reference zone=gen_submit and gen_conn. Zones
# can only be declared in the http context, not in a site file, and nginx fails
# to start with "unknown limit_req zone" if they are missing. This section was
# absent from earlier copies of this reference, which made the config below
# undeployable as written -- put these in the http block BEFORE the sites load.
limit_req_zone  $binary_remote_addr zone=gen_submit:10m rate=6r/m;
limit_conn_zone $binary_remote_addr zone=gen_conn:10m;

############ NGINX SITE CONFIG (/etc/nginx/sites-available) ############
server {
    listen 80 default_server;
    listen 443 ssl default_server;
    include snippets/ssl-origin.conf;
    server_name spbdatascience.org www.spbdatascience.org;
    root /var/www/spb-club/portfolio;
    index index.html;
    gzip on;
    gzip_types text/css text/javascript application/javascript application/json image/svg+xml;
    gzip_min_length 1024;

    # Security: never serve VCS metadata, dotfiles, OS junk, ops scripts or backups
    location ~ /\. { deny all; access_log off; log_not_found off; return 404; }
    location ~* \.(sh|conf|bak|backup)$ { deny all; return 404; }
    location ~* \.backup\.(html|css|js)$ { deny all; return 404; }

    # Versioned static assets (cache-busted via ?v=) can cache hard.
    location ~* \.(?:css|js|png|jpe?g|gif|svg|ico|webp|woff2?)$ {
        try_files $uri =404;
        add_header Cache-Control "public, max-age=31536000";
        include /etc/nginx/snippets/sec-headers.conf;
    }
    # DO NOT reinstate a 301 here. This line used to read
    #   location = /projects { return 301 https://demo.spbdatascience.org; }
    # and it is the single worst bug this site has had: 301 is permanent and
    # browsers cache it forever, so long after the rule was deleted, any visitor
    # who had ever loaded /projects still bounced to demo., with no server-side
    # trace, because the request never reached the origin. /projects is now a
    # real page. If a path may ever come back, redirect it 302, never 301.
    # See handoff.md, "Redirect hygiene".

    # HTML pages must always revalidate so a deploy is reflected immediately
    # (prevents stale pages loading an old gpu-status.js, etc.).
    location / {
        try_files $uri $uri/ $uri.html =404;
        add_header Cache-Control "no-cache";
        include /etc/nginx/snippets/sec-headers.conf;
    }
    location /api/gpu-status {
        proxy_pass http://127.0.0.1:15006/api/gpu-status;
        proxy_set_header Host $host;
        add_header Access-Control-Allow-Origin "*";
        include /etc/nginx/snippets/sec-headers.conf;
    }
}

# GPU worker access. GPU_WORKER_URL is where the worker lives; WORKER_TOKEN is the
# shared bearer token it requires on every request. Same token value in all five
# files. Generate with:  python3 -c "import secrets; print(secrets.token_urlsafe(32))"
#
# The token exists because the network is not the security boundary. In July 2026
# one stolen SSH key reached the private network and that was equivalent to full
# control of the GPU box. Anything that can route to :15100 must still hold the
# token before it can queue GPU work.
