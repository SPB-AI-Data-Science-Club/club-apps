############ NGINX SITE CONFIG (/etc/nginx/sites-available) ############
server {
    listen 80 default_server;
    listen 443 ssl default_server;
    include snippets/ssl-origin.conf;
    server_name spbdatascience.org www.spbdatascience.org;
    root /var/www/spb-club/portfolio;
    index index.html;
    gzip on;
    gzip_types text/css text/javascript application/javascript application/json image/svg+xml;
    gzip_min_length 1024;

    # Security: never serve VCS metadata, dotfiles, OS junk, ops scripts or backups
    location ~ /\. { deny all; access_log off; log_not_found off; return 404; }
    location ~* \.(sh|conf|bak|backup)$ { deny all; return 404; }
    location ~* \.backup\.(html|css|js)$ { deny all; return 404; }

    # Versioned static assets (cache-busted via ?v=) can cache hard.
    location ~* \.(?:css|js|png|jpe?g|gif|svg|ico|webp|woff2?)$ {
        try_files $uri =404;
        add_header Cache-Control "public, max-age=31536000";
        include /etc/nginx/snippets/sec-headers.conf;
    }
    # Old static /projects page retired; send it to the live demo showcase.
    location = /projects { return 301 https://demo.spbdatascience.org; }

    # HTML pages must always revalidate so a deploy is reflected immediately
    # (prevents stale pages loading an old gpu-status.js, etc.).
    location / {
        try_files $uri $uri/ $uri.html =404;
        add_header Cache-Control "no-cache";
        include /etc/nginx/snippets/sec-headers.conf;
    }
    location /api/gpu-status {
        proxy_pass http://127.0.0.1:15006/api/gpu-status;
        proxy_set_header Host $host;
        add_header Access-Control-Allow-Origin "*";
        include /etc/nginx/snippets/sec-headers.conf;
    }
}
